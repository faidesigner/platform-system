```js
// products/homepage/scripts/lib/csv.mjs
/**
 * RFC4180 CSV 파서. 인용 필드 내부의 개행·콤마·이스케이프된 따옴표를 보존한다.
 * 구글 시트 CSV export는 셀 안 줄바꿈을 인용 필드로 내보내므로 split(",") 사용 금지.
 */
export function parseCsv(text) {
  const rows = [];
  let row = [];
  let field = "";
  let quoted = false;
  let i = 0;
  const src = text.replace(/^﻿/, ""); // BOM 제거

  const endField = () => {
    row.push(field);
    field = "";
  };
  const endRow = () => {
    endField();
    rows.push(row);
    row = [];
  };

  while (i < src.length) {
    const ch = src[i];
    if (quoted) {
      if (ch === '"') {
        if (src[i + 1] === '"') {
          field += '"';
          i += 2;
          continue;
        }
        quoted = false;
        i += 1;
        continue;
      }
      field += ch;
      i += 1;
      continue;
    }
    if (ch === '"') {
      quoted = true;
      i += 1;
      continue;
    }
    if (ch === ",") {
      endField();
      i += 1;
      continue;
    }
    if (ch === "\r") {
      i += 1;
      continue;
    }
    if (ch === "\n") {
      endRow();
      i += 1;
      continue;
    }
    field += ch;
    i += 1;
  }
  if (field !== "" || row.length) endRow();
  return rows;
}

/** 1행을 헤더로 삼아 레코드 배열로 바꾼다. 헤더보다 짧은 행은 빈 문자열로 채운다. */
export function parseCsvRecords(text) {
  const [header, ...rest] = parseCsv(text);
  if (!header) return [];
  return rest.map((cells) =>
    Object.fromEntries(header.map((h, idx) => [h, cells[idx] ?? ""])),
  );
}
```

```js
/**
 * 번역 시트 ↔ messages 대조 로직(순수 함수). 네트워크·파일 IO 없음.
 * 시트는 단일 진실원천이 아니다 — 분류 결과를 사람이 판단하는 것을 전제로 한다.
 */
const SMART_QUOTES = [
  ["‘", "'"],
  ["’", "'"],
  ["“", '"'],
  ["”", '"'],
  [" ", " "],
];

export function normalize(value) {
  let s = String(value ?? "").normalize("NFC");
  for (const [from, to] of SMART_QUOTES) s = s.split(from).join(to);
  return s.replace(/\s+/g, " ").trim();
}

/** 줄바꿈·공백 차이를 무시한 동일성. 시트의 줄바꿈 지시를 판별하는 데 쓴다. */
export function sameIgnoringWhitespace(a, b) {
  const strip = (v) => String(v ?? "").normalize("NFC").replace(/\s+/g, "");
  return strip(a) === strip(b);
}

export function flattenMessages(tree, prefix = "") {
  const out = {};
  const entries = Array.isArray(tree)
    ? tree.map((v, i) => [String(i), v])
    : Object.entries(tree);
  for (const [key, value] of entries) {
    const path = prefix ? `${prefix}.${key}` : key;
    if (value !== null && typeof value === "object") Object.assign(out, flattenMessages(value, path));
    else out[path] = value;
  }
  return out;
}

export function buildKoIndex(koFlat) {
  const index = new Map();
  for (const [key, value] of Object.entries(koFlat)) {
    const k = normalize(value);
    index.set(k, [...(index.get(k) ?? []), key]);
  }
  return index;
}
```

```js
/**
 * 번역값이 아니라 "이 로케일에서는 숨겨라" 같은 지시가 적힌 셀.
 * 그대로 반영하면 화면에 한국어 지시문이 노출된다. 실제 사례: nav.careers, contact.toast.text.
 */
const INSTRUCTION_PATTERNS = [/미노출/, /제거/, /노출되는 영역/, /^번역x$/, /^\*/];

export function isInstructionCell(value) {
  const v = normalize(value);
  if (!v) return false;
  return INSTRUCTION_PATTERNS.some((re) => re.test(v));
}

/** sheet-decisions.json 배열 → `key locale` 조회용 Set */
export function loadDecisionSet(decisions) {
  return new Set((decisions ?? []).map((d) => `${d.key} ${d.locale}`));
}

export function classify({ key, locale, codeValue, sheetValue, ambiguous, decisions }) {
  if (isInstructionCell(sheetValue)) return "INSTRUCTION";
  if (decisions?.has(`${key} ${locale}`)) return "DECIDED";
  if (sameIgnoringWhitespace(codeValue, sheetValue)) return "WHITESPACE";
  if (ambiguous) return "AMBIGUOUS";
  return "CONTENT";
}
```

```js
// 지시문 셀도 리포트에는 남긴다(어느 티켓에서 처리됐는지 추적하려면 보여야 한다).
// 걸러내는 책임은 classify의 INSTRUCTION 버킷에 있다.
function pushDiffs({ out, keys, locale, sheetValue, messages, decisions, rowId, tab }) {
  const ambiguous = keys.length > 1;
  for (const key of keys) {
    const codeValue = messages[locale]?.[key];
    if (codeValue === undefined) continue;
    if (normalize(codeValue) === normalize(sheetValue)) continue;
    out.push({
      key,
      locale,
      codeValue,
      sheetValue,
      rowId,
      tab,
      ambiguous,
      bucket: classify({ key, locale, codeValue, sheetValue, ambiguous, decisions }),
    });
  }
}

/** IA 탭(nav/footer/main/product/aboutUs/media/contactUs) 대조 */
export function diffIaRows({ rows, messages, decisions, tab = "" }) {
  const koIndex = buildKoIndex(messages.ko);
  const out = [];
  for (const row of rows) {
    const ko = normalize(row.ko);
    if (!ko || ko === "번역x") continue;
    const keys = koIndex.get(ko);
    if (!keys) continue; // ko 미매칭 행은 별도 리포트(Task 5)에서 다룬다
    const rowId = `${row.IA ?? ""}/${row.screen ?? ""}/${row.property ?? ""}/${row.key ?? ""}`;
    for (const locale of LOCALES) {
      const sheetValue = row[locale];
      if (!normalize(sheetValue)) continue;
      pushDiffs({ out, keys, locale, sheetValue, messages, decisions, rowId, tab });
    }
  }
  return out;
}

    const isTitleLeaf = key.endsWith(".title") || /letterTitles\.\d+$/.test(key);
    if (!isTitleLeaf) continue;
    byKoTitle.set(normalize(value), key.replace(/\.title$/, ""));
  }

  const out = [];
  for (const row of rows) {
    const base = byKoTitle.get(normalize(row.ko_타이틀));
    if (!base) continue;
    const isLetter = base.includes("retailTechLetter");
    const rowId = `${row.contents ?? ""}/${normalize(row.ko_타이틀).slice(0, 24)}`;
    for (const locale of LOCALES) {
      const fields = isLetter
        ? [[base, row[`${locale}_타이틀`]]]
        : [
            [`${base}.title`, row[`${locale}_타이틀`]],
            [`${base}.description`, row[`${locale}_디스크립션`]],
          ];
      for (const [key, sheetValue] of fields) {
        if (!normalize(sheetValue)) continue;
        pushDiffs({ out, keys: [key], locale, sheetValue, messages, decisions, rowId, tab: "media" });
      }
    }
  }
  return out;
}

export function summarize(diffs) {
  return diffs.reduce((acc, d) => ({ ...acc, [d.bucket]: (acc[d.bucket] ?? 0) + 1 }), {});
}
```

```js
#!/usr/bin/env node
/**
 * 번역 시트 ↔ messages 대조 리포트 생성 (HOM-75).
 *
 * 시트는 링크 공개 상태라 인증 없이 CSV export가 된다.
 *   node scripts/sync-messages.mjs                 # 시트에서 받아 대조
 *   node scripts/sync-messages.mjs --fixture tmp/  # 로컬 CSV(csv_<gid>.csv)로 대조
 *
 * 이 스크립트는 messages를 **수정하지 않는다.** 리포트만 만든다 —
 * 시트가 단일 진실원천이 아니기 때문에(코드 확정안·ja 콘텐츠 변형) 반영은 사람이 판단한다.

import { mkdir, readFile, writeFile } from "node:fs/promises";
import path from "node:path";

  diffIaRows,
  diffMediaRows,
  flattenMessages,
  loadDecisionSet,
  normalize,

  { gid: "1826354096", name: "nav", kind: "ia" },

  { gid: "756952403", name: "product-vco", kind: "ia" },
  { gid: "758513153", name: "product-store", kind: "ia" },
  { gid: "800210217", name: "aboutUs", kind: "ia" },
  { gid: "2084621727", name: "media", kind: "ia" },
  { gid: "3671350", name: "contactUs", kind: "ia" },
  { gid: "2103530687", name: "youtube-news-letter", kind: "media" },
];
const BUCKET_LABEL = {
  CONTENT: "A. 문구 반영 검토 대상",
  AMBIGUOUS: "B. ko 원문 중복매칭 — 키 확인 필요",

const fixtureDir = fixtureIdx > -1 ? process.argv[fixtureIdx + 1] : null;

async function loadTab({ gid }) {
  if (fixtureDir) return readFile(path.join(fixtureDir, `csv_${gid}.csv`), "utf8");
  const url = `https://docs.google.com/spreadsheets/d/${SHEET_ID}/export?format=csv&gid=${gid}`;
  const res = await fetch(url, { redirect: "follow" });
  if (!res.ok) throw new Error(`시트 탭 ${gid} 취득 실패: HTTP ${res.status}`);
  return res.text();
}

async function loadMessages() {
  const read = async (locale) =>
    flattenMessages(JSON.parse(await readFile(path.resolve("messages", `${locale}.json`), "utf8")));
  return { ko: await read("ko"), en: await read("en"), ja: await read("ja") };
}

function renderReport({ diffs, unmatched, stamp }) {
  const counts = summarize(diffs);
  const lines = [
    `# 번역 시트 ↔ messages 대조 리포트 — ${stamp}`,
    "",
    "> `node scripts/sync-messages.mjs` 산출물. 시트는 단일 진실원천이 아니다 — 버킷 C·D는 반영 대상이 아니다.",
    "",
    "| 버킷 | 건수 |",
    "|---|---|",
    ...Object.keys(BUCKET_LABEL).map((b) => `| ${BUCKET_LABEL[b]} | ${counts[b] ?? 0} |`),
    `| ko 미매칭 행 | ${unmatched.length} |`,
    "",
  ];
  for (const bucket of Object.keys(BUCKET_LABEL)) {
    const rows = diffs.filter((d) => d.bucket === bucket);
    if (!rows.length) continue;
    lines.push(`## ${BUCKET_LABEL[bucket]} — ${rows.length}건`, "");
    lines.push("| loc | key | 코드 | 시트 | 행 |", "|---|---|---|---|---|");
    const cell = (v) => String(v).replace(/\|/g, "\\|").replace(/\n/g, " ⏎ ");
    for (const d of rows.sort((a, b) => a.key.localeCompare(b.key))) {
      lines.push(`| ${d.locale} | \`${d.key}\` | ${cell(d.codeValue)} | ${cell(d.sheetValue)} | ${d.rowId} |`);
    }
    lines.push("");
  }
  lines.push(`## ko 미매칭 행 — ${unmatched.length}건`, "");
  lines.push("> 고유명사·숫자·시트 전용 지시행이거나, 코드가 키를 쪼개 쓰는 행(줄바꿈 분할)이다.", "");
  lines.push("| 탭 | 행 | ko |", "|---|---|---|");
  for (const u of unmatched) {
    lines.push(`| ${u.tab} | ${u.rowId} | ${String(u.ko).replace(/\|/g, "\\|").replace(/\n/g, " ⏎ ")} |`);
  }
  return `${lines.join("\n")}\n`;
}

async function run() {
  const messages = await loadMessages();
  const decisionsFile = JSON.parse(await readFile(path.resolve("i18n/sheet-decisions.json"), "utf8"));
  const decisions = loadDecisionSet(decisionsFile.decisions);

  const diffs = [];
  const unmatched = [];
  const koSet = new Set(Object.values(messages.ko).map((v) => normalize(v)));

  for (const tab of TABS) {
    const rows = parseCsvRecords(await loadTab(tab));
    if (tab.kind === "ia") {
      diffs.push(...diffIaRows({ rows, messages, decisions, tab: tab.name }));
      for (const row of rows) {
        const ko = normalize(row.ko);
        if (!ko || ko === "번역x" || koSet.has(ko)) continue;
        unmatched.push({
          tab: tab.name,
          rowId: `${row.IA ?? ""}/${row.screen ?? ""}/${row.property ?? ""}/${row.key ?? ""}`,
          ko: row.ko,
        });
      }
    } else {
      diffs.push(...diffMediaRows({ rows, messages, decisions }));
    }
  }

  const stamp = new Date().toISOString().slice(0, 10).replace(/-/g, "");
  const out = path.resolve("docs", `HOM75_diff_${stamp}.md`);
  await mkdir(path.dirname(out), { recursive: true });
  await writeFile(out, renderReport({ diffs, unmatched, stamp }));

  console.log(`✓ ${diffs.length}건 diff → ${path.relative(process.cwd(), out)}`);
  console.table(summarize(diffs));
  console.log(`  ko 미매칭 행 ${unmatched.length}건`);
  console.log("\n다음: 버킷 A·B·E를 트리아지하고, 코드가 정답인 항목은 i18n/sheet-decisions.json에 추가하세요.");
}

run().catch((e) => {
  console.error("대조 실패:", e.message || e);
  process.exit(1);
});
```

- [x] **Step 1: 실패하는 테스트 작성**

```ts
// products/homepage/i18n/sheetDecisions.test.ts
import { describe, expect, it } from "vitest";

import decisionsFile from "@/i18n/sheet-decisions.json";
import en from "@/messages/en.json";
import ja from "@/messages/ja.json";

type Tree = { [key: string]: string | Tree };

function flatten(tree: Tree, prefix = ""): Record<string, string> {
  return Object.entries(tree).reduce<Record<string, string>>((acc, [key, value]) => {
    const path = prefix ? `${prefix}.${key}` : key;
    if (typeof value === "string") acc[path] = value;
    else Object.assign(acc, flatten(value, path));
    return acc;
  }, {});
}

const MESSAGES: Record<string, Record<string, string>> = {
  en: flatten(en as Tree),
  ja: flatten(ja as Tree),
};

describe("sheet-decisions.json", () => {
  // 선언된 codeValue가 실제 messages와 어긋나면, 대조 도구가 실제 이격을 DECIDED로 숨긴다.
  it("모든 선언의 codeValue가 현재 messages 값과 일치한다", () => {
    const stale = decisionsFile.decisions
      .filter((d) => MESSAGES[d.locale]?.[d.key] !== d.codeValue)
      .map((d) => `${d.key}(${d.locale}): 선언=${d.codeValue} / 실제=${MESSAGES[d.locale]?.[d.key]}`);
    expect(stale).toEqual([]);
  });

  it("모든 선언에 reason과 decidedAt이 있다", () => {
    const missing = decisionsFile.decisions
      .filter((d) => !d.reason?.trim() || !/^\d{4}-\d{2}-\d{2}$/.test(d.decidedAt ?? ""))
      .map((d) => `${d.key}(${d.locale})`);
    expect(missing).toEqual([]);
  });

  it("같은 (key, locale)을 중복 선언하지 않는다", () => {
    const seen = decisionsFile.decisions.map((d) => `${d.key}|${d.locale}`);
    expect(seen.length).toBe(new Set(seen).size);
  });
});
```
